const DEFAULT_EXPENSE_CATEGORIES = ['อาหาร', 'เดินทาง', 'ที่พัก/บ้าน', 'ช้อปปิ้ง', 'สุขภาพ', 'บันเทิง', 'ครอบครัว', 'บิล/ค่าใช้จ่ายประจำ', 'การศึกษา', 'อื่นๆ'];
const INCOME_CATEGORIES = ['เงินเดือน', 'รายได้เสริม', 'อื่นๆ'];

const TYPE_LABEL = { income: 'รายรับ', personal_expense: 'รายจ่ายส่วนตัว', pending_claim: 'รอเบิก/รอบิล', investment: 'ลงทุน', debt_payment: 'ชำระหนี้', loan_given: 'ให้ยืมเงิน' };
const CLAIM_STATUS_LABEL = { awaiting_bill: 'รอบิล', pending: 'รอตั้งเบิก', claimed: 'ตั้งเบิกแล้ว', received: 'ได้รับคืนแล้ว' };
const CLAIM_STATUS_ORDER = ['awaiting_bill', 'pending', 'claimed', 'received'];

function uid() { return Date.now().toString(36) + Math.random().toString(36).slice(2, 8); }
function todayStr() { return new Date().toISOString().slice(0, 10); }
function fmtMoney(n) { return (n < 0 ? '-' : '') + Math.abs(n || 0).toLocaleString('th-TH', { minimumFractionDigits: 0, maximumFractionDigits: 2 }); }
function monthKey(d) { return d.slice(0, 7); }
function monthLabel(key) {
  const [y, m] = key.split('-');
  const names = ['ม.ค.', 'ก.พ.', 'มี.ค.', 'เม.ย.', 'พ.ค.', 'มิ.ย.', 'ก.ค.', 'ส.ค.', 'ก.ย.', 'ต.ค.', 'พ.ย.', 'ธ.ค.'];
  return `${names[parseInt(m, 10) - 1]} ${(parseInt(y, 10) + 543).toString().slice(-2)}`;
}
function cashDelta(t, categoryStatusMap) {
  if (t.type === 'income') return t.amount;
  if (t.type === 'personal_expense') {
    const status = t.category ? (categoryStatusMap || {})[t.category] : '';
    if (status && t.claimReceived) return 0;
    return -t.amount;
  }
  if (t.type === 'investment') return t.direction === 'in' ? -t.amount : t.amount;
  if (t.type === 'pending_claim') return t.status === 'received' ? 0 : -t.amount;
  if (t.type === 'debt_payment') return -t.amount;
  if (t.type === 'loan_given') return t.status === 'repaid' ? 0 : -t.amount;
  return 0;
}
function paymentLabel(pm, cards, cashAccounts) {
  if (!pm || pm === 'cash') return 'เงินสด/โอน';
  const acc = (cashAccounts || []).find(a => a.id === pm);
  if (acc) return acc.name;
  const c = (cards || []).find(c => c.id === pm);
  return c ? c.name : 'ไม่ระบุ';
}
function isCardId(pm, cards) { return (cards || []).some(c => c.id === pm); }

async function aiCategorize(description, categoryNames) {
  if (!description || !description.trim() || !categoryNames || categoryNames.length === 0) return null;
  try {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: 'claude-sonnet-4-6',
        max_tokens: 1000,
        messages: [{
          role: 'user',
          content: `จัดหมวดหมู่รายการค่าใช้จ่ายนี้: "${description}"\n\nเลือกได้เฉพาะหมวดหมู่ต่อไปนี้เท่านั้น: ${categoryNames.join(', ')}\n\nตอบกลับด้วยชื่อหมวดหมู่คำเดียวเป๊ะๆ ตามที่ให้มา ห้ามมีคำอธิบายอื่นใด`
        }]
      })
    });
    const data = await res.json();
    const text = (data.content || []).find(b => b.type === 'text')?.text?.trim();
    return categoryNames.includes(text) ? text : null;
  } catch (e) {
    return null;
  }
}

const emptyForm = {
  type: 'personal_expense', amount: '', description: '', date: todayStr(),
  category: '', customCategory: '', paymentMethod: 'cash', direction: 'in', buyer: 'self',
  awaitingBill: false, claimCategory: '', debtId: ''
};

export default function FinanceTracker() {
  const [transactions, setTransactions] = useState(null);
  const [cards, setCards] = useState([]);
  const [cashAccounts, setCashAccounts] = useState([]);
  const [expenseCategories, setExpenseCategories] = useState(DEFAULT_EXPENSE_CATEGORIES.map(name => ({ name, status: '' })));
  const [claimCategories, setClaimCategories] = useState([]);
  const [debts, setDebts] = useState([]);
  const [tab, setTab] = useState('dashboard');
  const [form, setForm] = useState(emptyForm);
  const [reportMonth, setReportMonth] = useState(monthKey(todayStr()));
  const [saveErr, setSaveErr] = useState('');
  const [newCardName, setNewCardName] = useState('');
  const [newCashName, setNewCashName] = useState('');
  const [newDebtName, setNewDebtName] = useState('');
  const [newDebtAmt, setNewDebtAmt] = useState('');
  const [newExpCatName, setNewExpCatName] = useState('');
  const [newExpCatStatus, setNewExpCatStatus] = useState('');
  const [selectedTx, setSelectedTx] = useState(null);
  const [viewCategory, setViewCategory] = useState(null);
  const [aiBusyIds, setAiBusyIds] = useState([]);
  const [bulkAiBusy, setBulkAiBusy] = useState(false);
  const [toast, setToast] = useState(null);
  const toastTimer = useRef(null);

  function showToast(message, undoId) {
    setToast({ message, undoId });
    clearTimeout(toastTimer.current);
    toastTimer.current = setTimeout(() => setToast(null), 6000);
  }

  useEffect(() => {
    (async () => {
      try {
        const r = await window.storage.get('app-data', false);
        const parsed = r ? JSON.parse(r.value) : {};
        const cutoff = Date.now() - 7 * 24 * 60 * 60 * 1000;
        const rawTx = parsed.transactions || [];
        const cleaned = rawTx.filter(t => !(t.deletedAt && t.deletedAt < cutoff));
        setTransactions(cleaned);
        setCards(parsed.cards || []);
        setCashAccounts(parsed.cashAccounts || []);
        const rawCats = parsed.expenseCategories || DEFAULT_EXPENSE_CATEGORIES.map(name => ({ name, status: '' }));
        setExpenseCategories(rawCats.map(c => typeof c === 'string' ? { name: c, status: '' } : c));
        setClaimCategories(parsed.claimCategories || []);
        setDebts(parsed.debts || []);
        if (cleaned.length !== rawTx.length) {
          window.storage.set('app-data', JSON.stringify({ ...parsed, transactions: cleaned }), false).catch(() => {});
        }
      } catch (e) {
        setTransactions([]); setCards([]); setCashAccounts([]);
        setExpenseCategories(DEFAULT_EXPENSE_CATEGORIES.map(name => ({ name, status: '' })));
        setClaimCategories([]); setDebts([]);
      }
    })();
  }, []);

  async function persist(updates) {
    const next = {
      transactions: updates.transactions !== undefined ? updates.transactions : transactions,
      cards: updates.cards !== undefined ? updates.cards : cards,
      cashAccounts: updates.cashAccounts !== undefined ? updates.cashAccounts : cashAccounts,
      expenseCategories: updates.expenseCategories !== undefined ? updates.expenseCategories : expenseCategories,
      claimCategories: updates.claimCategories !== undefined ? updates.claimCategories : claimCategories,
      debts: updates.debts !== undefined ? updates.debts : debts,
    };
    if (updates.transactions !== undefined) setTransactions(updates.transactions);
    if (updates.cards !== undefined) setCards(updates.cards);
    if (updates.cashAccounts !== undefined) setCashAccounts(updates.cashAccounts);
    if (updates.expenseCategories !== undefined) setExpenseCategories(updates.expenseCategories);
    if (updates.claimCategories !== undefined) setClaimCategories(updates.claimCategories);
    if (updates.debts !== undefined) setDebts(updates.debts);
    try {
      const ok = await window.storage.set('app-data', JSON.stringify(next), false);
      if (!ok) setSaveErr('บันทึกไม่สำเร็จ ลองอีกครั้ง'); else setSaveErr('');
    } catch (e) {
      setSaveErr('บันทึกไม่สำเร็จ ลองอีกครั้ง');
    }
  }

  const categoryStatusMap = useMemo(() => Object.fromEntries(expenseCategories.map(c => [c.name, c.status || ''])), [expenseCategories]);
  const liveTx = useMemo(() => (transactions || []).filter(t => !t.deletedAt), [transactions]);
  const trashList = useMemo(() => (transactions || []).filter(t => t.deletedAt).sort((a, b) => b.deletedAt - a.deletedAt), [transactions]);
  const available = useMemo(() => liveTx.reduce((s, t) => s + cashDelta(t, categoryStatusMap), 0), [liveTx, categoryStatusMap]);
  const pendingList = useMemo(() => liveTx.filter(t => t.type === 'pending_claim' && t.status !== 'received').sort((a, b) => a.date < b.date ? 1 : -1), [liveTx]);
  const pendingTotal = useMemo(() => pendingList.reduce((s, t) => s + t.amount, 0), [pendingList]);
  const folderPendingList = useMemo(() => liveTx.filter(t => t.type === 'personal_expense' && t.category && categoryStatusMap[t.category] && !t.claimReceived), [liveTx, categoryStatusMap]);
  const folderPendingTotal = useMemo(() => folderPendingList.reduce((s, t) => s + t.amount, 0), [folderPendingList]);
  const folderPendingGrouped = useMemo(() => {
    const groups = {};
    folderPendingList.forEach(t => { (groups[t.category] = groups[t.category] || []).push(t); });
    return Object.entries(groups).map(([name, items]) => ({ name, status: categoryStatusMap[name], items, total: items.reduce((s, t) => s + t.amount, 0) }));
  }, [folderPendingList, categoryStatusMap]);
  const loanList = useMemo(() => liveTx.filter(t => t.type === 'loan_given' && t.status !== 'repaid').sort((a, b) => a.date < b.date ? 1 : -1), [liveTx]);
  const loanTotal = useMemo(() => loanList.reduce((s, t) => s + t.amount, 0), [loanList]);
  const combinedPendingTotal = pendingTotal + folderPendingTotal;
  const statusTotals = useMemo(() => {
    const m = {};
    folderPendingList.forEach(t => { const s = categoryStatusMap[t.category] || 'ไม่ระบุสถานะ'; m[s] = (m[s] || 0) + t.amount; });
    pendingList.forEach(t => { const s = CLAIM_STATUS_LABEL[t.status]; m[s] = (m[s] || 0) + t.amount; });
    return Object.entries(m).map(([label, total]) => ({ label, total })).sort((a, b) => b.total - a.total);
  }, [folderPendingList, pendingList, categoryStatusMap]);
  const debtTotal = useMemo(() => debts.reduce((s, d) => s + (d.balance || 0), 0), [debts]);
  const unsortedList = useMemo(() => liveTx.filter(t => t.type === 'personal_expense' && !t.category), [liveTx]);
  const recent = useMemo(() => [...liveTx].sort((a, b) => (b.date + b.id) < (a.date + a.id) ? -1 : 1).slice(0, 6), [liveTx]);
  const categoryCounts = useMemo(() => {
    const m = {};
    liveTx.filter(t => t.type === 'personal_expense' && t.category).forEach(t => { m[t.category] = (m[t.category] || 0) + 1; });
    return m;
  }, [liveTx]);
  const pendingGrouped = useMemo(() => {
    const groups = {};
    pendingList.forEach(t => { const k = t.claimCategory || '__none__'; (groups[k] = groups[k] || []).push(t); });
    const orderedKeys = [...claimCategories.filter(c => groups[c]), ...(groups['__none__'] ? ['__none__'] : [])];
    return orderedKeys.map(k => ({ key: k, label: k === '__none__' ? 'ไม่ระบุหมวด' : k, items: groups[k], total: groups[k].reduce((s, t) => s + t.amount, 0) }));
  }, [pendingList, claimCategories]);

  function resetForm(type) { setForm({ ...emptyForm, type, date: todayStr() }); }

  function submit() {
    const amt = parseFloat(form.amount);
    if (!amt || amt <= 0) { setSaveErr('กรุณาใส่จำนวนเงินให้ถูกต้อง'); return; }
    if (!form.date) { setSaveErr('กรุณาเลือกวันที่'); return; }
    if (form.type === 'debt_payment' && !form.debtId) { setSaveErr('กรุณาเลือกหนี้ที่จะชำระ'); return; }
    setSaveErr('');
    const t = { id: uid(), type: form.type, amount: amt, description: form.description.trim(), date: form.date };
    let cardTouched = null;
    if (form.type === 'personal_expense') {
      t.category = null;
      t.paymentMethod = form.paymentMethod;
      if (isCardId(form.paymentMethod, cards)) cardTouched = form.paymentMethod;
    } else if (form.type === 'income') {
      t.category = form.category === 'อื่นๆ' && form.customCategory ? form.customCategory.trim() : (form.category || 'อื่นๆ');
    } else if (form.type === 'pending_claim') {
      t.buyer = form.buyer; t.status = form.awaitingBill ? 'awaiting_bill' : 'pending'; t.paymentMethod = form.paymentMethod;
      t.claimCategory = form.claimCategory || null;
      if (isCardId(form.paymentMethod, cards)) cardTouched = form.paymentMethod;
    } else if (form.type === 'investment') {
      t.direction = form.direction;
    } else if (form.type === 'debt_payment') {
      t.debtId = form.debtId;
    } else if (form.type === 'loan_given') {
      t.status = 'outstanding';
    }
    const nextTx = [...(transactions || []), t];
    const updates = { transactions: nextTx };
    if (cardTouched) updates.cards = cards.map(c => c.id === cardTouched ? { ...c, balance: (c.balance || 0) + amt } : c);
    if (form.type === 'debt_payment') updates.debts = debts.map(d => d.id === form.debtId ? { ...d, balance: Math.max(0, (d.balance || 0) - amt) } : d);
    persist(updates);
    resetForm(form.type);
    setTab('dashboard');
  }

  function updateClaimStatus(id, status) { persist({ transactions: (transactions || []).map(t => t.id === id ? { ...t, status } : t) }); }
  function updateLoanStatus(id) { persist({ transactions: (transactions || []).map(t => t.id === id ? { ...t, status: 'repaid' } : t) }); }
  function updateFolderClaim(id) { persist({ transactions: (transactions || []).map(t => t.id === id ? { ...t, claimReceived: true } : t) }); }

  function removeTx(id) {
    const t = (transactions || []).find(x => x.id === id);
    if (!t || t.deletedAt) return;
    const nextTx = (transactions || []).map(x => x.id === id ? { ...x, deletedAt: Date.now() } : x);
    const updates = { transactions: nextTx };
    if ((t.type === 'personal_expense' || t.type === 'pending_claim') && isCardId(t.paymentMethod, cards)) {
      updates.cards = cards.map(c => c.id === t.paymentMethod ? { ...c, balance: Math.max(0, (c.balance || 0) - t.amount) } : c);
    }
    if (t.type === 'debt_payment' && t.debtId) {
      updates.debts = debts.map(d => d.id === t.debtId ? { ...d, balance: (d.balance || 0) + t.amount } : d);
    }
    persist(updates);
    showToast(`ลบ "${t.description || TYPE_LABEL[t.type]}" แล้ว — กู้คืนได้จากถังขยะถ้าเผลอกดผิด`, id);
  }
  function restoreTx(id) {
    const t = (transactions || []).find(x => x.id === id);
    if (!t) return;
    const nextTx = (transactions || []).map(x => x.id === id ? { ...x, deletedAt: undefined } : x);
    const updates = { transactions: nextTx };
    if ((t.type === 'personal_expense' || t.type === 'pending_claim') && isCardId(t.paymentMethod, cards)) {
      updates.cards = cards.map(c => c.id === t.paymentMethod ? { ...c, balance: (c.balance || 0) + t.amount } : c);
    }
    if (t.type === 'debt_payment' && t.debtId) {
      updates.debts = debts.map(d => d.id === t.debtId ? { ...d, balance: Math.max(0, (d.balance || 0) - t.amount) } : d);
    }
    persist(updates);
    setToast(null);
  }
  function purgeTx(id) { persist({ transactions: (transactions || []).filter(x => x.id !== id) }); }
  function setCategory(id, category) { persist({ transactions: (transactions || []).map(t => t.id === id ? { ...t, category } : t) }); }

  function addCard() {
    const name = newCardName.trim();
    if (!name) return;
    persist({ cards: [...cards, { id: uid(), name, creditLimit: 0, balance: 0 }] });
    setNewCardName('');
  }
  function removeCard(id) { persist({ cards: cards.filter(c => c.id !== id) }); }
  function payCard(id) { persist({ cards: cards.map(c => c.id === id ? { ...c, balance: 0 } : c) }); }
  function updateCardLimit(id, val) { persist({ cards: cards.map(c => c.id === id ? { ...c, creditLimit: val } : c) }); }

  function addCashAccount() {
    const name = newCashName.trim();
    if (!name) return;
    persist({ cashAccounts: [...cashAccounts, { id: uid(), name }] });
    setNewCashName('');
  }
  function removeCashAccount(id) { persist({ cashAccounts: cashAccounts.filter(a => a.id !== id) }); }

  function addDebt() {
    const name = newDebtName.trim();
    if (!name) return;
    persist({ debts: [...debts, { id: uid(), name, balance: parseFloat(newDebtAmt) || 0 }] });
    setNewDebtName(''); setNewDebtAmt('');
  }
  function removeDebt(id) { persist({ debts: debts.filter(d => d.id !== id) }); }
  function updateDebtBalance(id, val) { persist({ debts: debts.map(d => d.id === id ? { ...d, balance: val } : d) }); }

  function addExpenseCategory() {
    const name = newExpCatName.trim();
    if (!name || expenseCategories.some(c => c.name === name)) return;
    persist({ expenseCategories: [...expenseCategories, { name, status: newExpCatStatus.trim() }] });
    setNewExpCatName(''); setNewExpCatStatus('');
  }
  function renameExpenseCategory(oldName, newName) {
    if (!newName || newName === oldName) return;
    const nextList = expenseCategories.map(c => c.name === oldName ? { ...c, name: newName } : c);
    const nextTx = (transactions || []).map(t => t.type === 'personal_expense' && t.category === oldName ? { ...t, category: newName } : t);
    persist({ transactions: nextTx, expenseCategories: nextList });
  }
  function updateExpenseCategoryStatus(name, status) {
    persist({ expenseCategories: expenseCategories.map(c => c.name === name ? { ...c, status } : c) });
  }
  function deleteExpenseCategory(name) {
    const nextList = expenseCategories.filter(c => c.name !== name);
    const nextTx = (transactions || []).map(t => t.type === 'personal_expense' && t.category === name ? { ...t, category: null } : t);
    persist({ transactions: nextTx, expenseCategories: nextList });
  }

  function addClaimCategory(name) {
    name = (name || '').trim();
    if (!name || claimCategories.includes(name)) return;
    persist({ claimCategories: [...claimCategories, name] });
  }
  function renameClaimCategory(oldName, newName) {
    if (!newName || newName === oldName) return;
    const nextList = claimCategories.map(c => c === oldName ? newName : c);
    const nextTx = (transactions || []).map(t => t.type === 'pending_claim' && t.claimCategory === oldName ? { ...t, claimCategory: newName } : t);
    persist({ transactions: nextTx, claimCategories: nextList });
  }
  function deleteClaimCategory(name) {
    const nextList = claimCategories.filter(c => c !== name);
    const nextTx = (transactions || []).map(t => t.type === 'pending_claim' && t.claimCategory === name ? { ...t, claimCategory: null } : t);
    persist({ transactions: nextTx, claimCategories: nextList });
  }

  function onCardDragStart(e, id) { e.dataTransfer.setData('text/plain', id); }
  function onColumnDrop(e, category) {
    e.preventDefault();
    const id = e.dataTransfer.getData('text/plain');
    if (id) setCategory(id, category);
  }
  function onColumnClick(category) {
    if (selectedTx) { setCategory(selectedTx, category); setSelectedTx(null); return; }
    setViewCategory(v => v === category ? null : category);
  }
  function toggleSelect(id) { setSelectedTx(sel => sel === id ? null : id); }

  async function aiAssistOne(t) {
    setAiBusyIds(ids => [...ids, t.id]);
    const cat = await aiCategorize(t.description, expenseCategories.map(c => c.name));
    setAiBusyIds(ids => ids.filter(i => i !== t.id));
    if (cat) setCategory(t.id, cat);
  }

  async function aiAssistAll() {
    const targets = unsortedList.filter(t => t.description && t.description.trim());
    if (targets.length === 0) return;
    setBulkAiBusy(true);
    const names = expenseCategories.map(c => c.name);
    const results = await Promise.all(targets.map(async t => ({ id: t.id, cat: await aiCategorize(t.description, names) })));
    setBulkAiBusy(false);
    const map = {};
    results.forEach(r => { if (r.cat) map[r.id] = r.cat; });
    persist({ transactions: (transactions || []).map(t => map[t.id] ? { ...t, category: map[t.id] } : t) });
  }

  const monthTx = useMemo(() => liveTx.filter(t => monthKey(t.date) === reportMonth), [liveTx, reportMonth]);
  const monthExpenseByCat = useMemo(() => {
    const m = {};
    monthTx.filter(t => t.type === 'personal_expense' && t.category).forEach(t => { m[t.category] = (m[t.category] || 0) + t.amount; });
    return Object.entries(m).map(([name, value]) => ({ name, value })).sort((a, b) => b.value - a.value);
  }, [monthTx]);
  const monthUnsortedTotal = useMemo(() => monthTx.filter(t => t.type === 'personal_expense' && !t.category).reduce((s, t) => s + t.amount, 0), [monthTx]);
  const monthIncome = useMemo(() => monthTx.filter(t => t.type === 'income').reduce((s, t) => s + t.amount, 0), [monthTx]);
  const monthExpense = useMemo(() => monthTx.filter(t => t.type === 'personal_expense').reduce((s, t) => s + t.amount, 0), [monthTx]);
  const trend = useMemo(() => {
    const now = new Date(reportMonth + '-01T00:00:00');
    const arr = [];
    for (let i = 5; i >= 0; i--) {
      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
      const key = d.toISOString().slice(0, 7);
      const tx = liveTx.filter(t => monthKey(t.date) === key);
      arr.push({
        key, name: monthLabel(key),
        รายรับ: tx.filter(t => t.type === 'income').reduce((s, t) => s + t.amount, 0),
        รายจ่าย: tx.filter(t => t.type === 'personal_expense').reduce((s, t) => s + t.amount, 0)
      });
    }
    return arr;
  }, [liveTx, reportMonth]);

  function shiftMonth(delta) {
    const [y, m] = reportMonth.split('-').map(Number);
    const d = new Date(y, m - 1 + delta, 1);
    setReportMonth(d.toISOString().slice(0, 7));
  }

  if (transactions === null) {
    return <div className="pft"><style>{CSS}</style><div className="pft-loading">กำลังโหลดข้อมูล...</div></div>;
  }

  const viewCategoryItems = viewCategory ? liveTx.filter(t => t.type === 'personal_expense' && t.category === viewCategory).sort((a, b) => b.date < a.date ? -1 : 1) : [];

            {monthTx.length === 0 ? <p className="pft-empty">ไม่มีรายการเดือนนี้</p> : (
              <ul className="pft-list">
                {[...monthTx].sort((a, b) => b.date < a.date ? -1 : 1).map(t => (
                  <li key={t.id} className="pft-histrow">
                    <div className="pft-tx-info">
                      <span className="pft-tx-desc">{t.description || TYPE_LABEL[t.type]}</span>
                      <span className="pft-tx-meta">{t.date} · {TYPE_LABEL[t.type]}{t.type === 'personal_expense' ? ` · ${paymentLabel(t.paymentMethod, cards, cashAccounts)}` : ''}</span>
                    </div>
                    {t.type === 'personal_expense' ? (
                      <select className="pft-histcat" value={t.category || ''} onChange={e => setCategory(t.id, e.target.value || null)}>
                        <option value="">ยังไม่จัด</option>
                        {expenseCategories.map(c => <option key={c.name} value={c.name}>{c.name}</option>)}
                      </select>
                    ) : <span className="pft-histcat pft-histcat-static">{t.category || t.claimCategory || CLAIM_STATUS_LABEL[t.status] || ''}</span>}
                    <span className={`pft-tx-amt ${t.type === 'income' || (t.type === 'investment' && t.direction === 'out') ? 'pft-green' : 'pft-red'}`}>฿{fmtMoney(t.amount)}</span>
                    <button className="pft-del" onClick={() => removeTx(t.id)}><Trash2 size={14} /></button>
                  </li>
                ))}
              </ul>
            )}
          </div>
        )}

        {tab === 'settings' && (
          <div className="pft-stack">
            <h2 className="pft-section-title" style={{ marginTop: 0 }}><Wallet size={16} style={{ verticalAlign: '-3px', marginRight: '6px' }} />บัญชีเงินสด/ธนาคาร</h2>
            <div className="pft-addcard">
              <input type="text" placeholder="เช่น เงินสดในกระเป๋า, บัญชีกสิกร" value={newCashName} onChange={e => setNewCashName(e.target.value)} onKeyDown={e => e.key === 'Enter' && addCashAccount()} />
              <button type="button" onClick={addCashAccount}>เพิ่ม</button>
            </div>
            {cashAccounts.length === 0 && <p className="pft-empty">ยังไม่มีบัญชี — ถ้าไม่เพิ่ม ระบบจะใช้ "เงินสด/โอน" แบบรวมแทน</p>}
            <ul className="pft-list">
              {cashAccounts.map(a => (
                <li key={a.id} className="pft-tx">

                </li>
              ))}

            <div className="pft-addcard">
              <input type="text" placeholder="ชื่อบัตร เช่น กสิกร แพลทินัม" value={newCardName} onChange={e => setNewCardName(e.target.value)} onKeyDown={e => e.key === 'Enter' && addCard()} />
              <button type="button" onClick={addCard}>เพิ่ม</button>
            </div>

            <ul className="pft-list">
              {cards.map(c => {

                      <CreditCard size={18} className="pft-blue" />
                      <span className="pft-cardrow-name">{c.name}</span>
                      <button className="pft-del" onClick={() => removeCard(c.id)}><Trash2 size={14} /></button>
                    </div>
                    {limit > 0 && <div className="pft-cardrow-bar"><div className="pft-cardrow-barfill" style={{ width: pct + '%' }} /></div>}
                    <div className="pft-cardrow-fields">
                      <span>ใช้ไปแล้ว <b>฿{fmtMoney(used)}</b>{limit > 0 ? ` / ฿${fmtMoney(limit)}` : ''}</span>

                    </div>

              <input type="text" placeholder="ชื่อหนี้ เช่น ผ่อนรถ" value={newDebtName} onChange={e => setNewDebtName(e.target.value)} />

              <button type="button" onClick={addDebt}>เพิ่ม</button>
            </div>
            {debts.length === 0 && <p className="pft-empty">ยังไม่มีหนี้ในระบบ</p>}

                  </div>
                </li>
              ))}
            </ul>

            <h2 className="pft-section-title"><LayoutGrid size={16} style={{ verticalAlign: '-3px', marginRight: '6px' }} />หมวดหมู่รายจ่ายส่วนตัว</h2>

            <div className="pft-addcat-row">
              <input type="text" placeholder="ชื่อหมวดใหม่" value={newExpCatName} onChange={e => setNewExpCatName(e.target.value)} />
              <input type="text" placeholder="สถานะ (ไม่บังคับ)" value={newExpCatStatus} onChange={e => setNewExpCatStatus(e.target.value)} onKeyDown={e => e.key === 'Enter' && addExpenseCategory()} />
              <button type="button" onClick={addExpenseCategory}>เพิ่ม</button>
            </div>
            {expenseCategories.length === 0 && <p className="pft-empty">ยังไม่มีหมวด</p>}
            <ul className="pft-list">
              {expenseCategories.map(c => (
                <li key={c.name} className="pft-catrow2">
                  <input className="pft-catrename" defaultValue={c.name}
                    onBlur={e => { const v = e.target.value.trim(); if (v && v !== c.name) renameExpenseCategory(c.name, v); else e.target.value = c.name; }}

                    onKeyDown={e => e.key === 'Enter' && e.target.blur()} />
                  <button className="pft-del" onClick={() => deleteExpenseCategory(c.name)}><Trash2 size={14} /></button>
                </li>
              ))}
            </ul>

            <h2 className="pft-section-title"><Clock size={16} style={{ verticalAlign: '-3px', marginRight: '6px' }} />หมวดรายการรอเบิก/รอบิล</h2>
            <p className="pft-hint">ใช้จัดกลุ่มรายการที่บันทึกแบบ "รายจ่ายรอเบิก" เต็มรูปแบบ (มีผู้สำรองจ่าย/สถานะ 4 ขั้น)</p>
            <CategoryEditor categories={claimCategories} placeholder="เพิ่มหมวดใหม่ เช่น วัสดุก่อสร้าง(รอบิล)"
              onAdd={addClaimCategory} onRename={renameClaimCategory} onDelete={deleteClaimCategory} />

            <h2 className="pft-section-title"><Trash2 size={16} style={{ verticalAlign: '-3px', marginRight: '6px' }} />ถังขยะ{trashList.length > 0 ? ` (${trashList.length})` : ''}</h2>
            {trashList.length === 0 ? <p className="pft-empty">ถังขยะว่าง</p> : (
              <>
                <p className="pft-hint">รายการในถังขยะจะถูกลบถาวรอัตโนมัติหลังผ่านไป 7 วัน</p>
                <ul className="pft-list">
                  {trashList.map(t => {
                    const daysLeft = Math.max(0, 7 - Math.floor((Date.now() - t.deletedAt) / (24 * 60 * 60 * 1000)));
                    return (
                      <li key={t.id} className="pft-trashrow">
                        <div className="pft-tx-info">
                          <span className="pft-tx-desc">{t.description || TYPE_LABEL[t.type]}</span>

                        </div>
                        <button type="button" className="pft-restorebtn" onClick={() => restoreTx(t.id)}>กู้คืน</button>
                        <button className="pft-del" onClick={() => purgeTx(t.id)}><Trash2 size={14} /></button>
                      </li>
                    );
                  })}
                </ul>
              </>
            )}
          </div>
        )}
      </main>

      {toast && (
        <div className="pft-toast">
          <span>{toast.message}</span>
          <button type="button" className="pft-toast-undo" onClick={() => restoreTx(toast.undoId)}>เลิกทำ</button>
          <button type="button" className="pft-toast-x" onClick={() => setToast(null)}><X size={14} /></button>
        </div>
      )}
    </div>
  );
}

function CategoryEditor({ categories, onRename, onDelete, onAdd, placeholder }) {
  const [draft, setDraft] = useState('');
  return (
    <>
      <div className="pft-addcard">
        <input type="text" placeholder={placeholder} value={draft} onChange={e => setDraft(e.target.value)}
          onKeyDown={e => { if (e.key === 'Enter') { onAdd(draft); setDraft(''); } }} />
        <button type="button" onClick={() => { onAdd(draft); setDraft(''); }}>เพิ่ม</button>
      </div>
      {categories.length === 0 && <p className="pft-empty">ยังไม่มีหมวด</p>}
      <ul className="pft-list">
        {categories.map(cat => (
          <li key={cat} className="pft-tx">
            <input className="pft-catrename" defaultValue={cat}
              onBlur={e => { const v = e.target.value.trim(); if (v && v !== cat) onRename(cat, v); else e.target.value = cat; }}
              onKeyDown={e => e.key === 'Enter' && e.target.blur()} />
            <button className="pft-del" onClick={() => onDelete(cat)}><Trash2 size={14} /></button>
          </li>
        ))}
      </ul>
    </>
  );
}

function PaymentPicker({ value, cards, cashAccounts, onChange, onManage }) {
  return (
    <div className="pft-paymentpicker">
      {cashAccounts.length === 0 && (
        <button type="button" className={value === 'cash' ? 'active' : ''} onClick={() => onChange('cash')}><Banknote size={14} /> เงินสด/โอน</button>
      )}
      {cashAccounts.map(a => (
        <button type="button" key={a.id} className={value === a.id ? 'active' : ''} onClick={() => onChange(a.id)}><Wallet size={14} /> {a.name}</button>
      ))}
      {cards.map(c => (
        <button type="button" key={c.id} className={value === c.id ? 'active' : ''} onClick={() => onChange(c.id)}><CreditCard size={14} /> {c.name}</button>
      ))}
      <button type="button" className="pft-managebtn" onClick={onManage}>+ จัดการบัญชี/บัตร</button>
    </div>
  );
}

function TxRow({ t, cards, cashAccounts, onDelete }) {
  const icon = t.type === 'income' ? <ArrowUpCircle size={18} className="pft-green" /> :
    t.type === 'personal_expense' ? <ArrowDownCircle size={18} className="pft-red" /> :
    t.type === 'pending_claim' ? <Landmark size={18} className="pft-amber" /> :
    t.type === 'debt_payment' ? <CreditCard size={18} className="pft-purple" /> :
    t.type === 'loan_given' ? <HandCoins size={18} className="pft-amber" /> :
    <TrendingUp size={18} className="pft-blue" />;
  const sign = t.type === 'income' ? '+' : t.type === 'investment' ? (t.direction === 'in' ? '-' : '+') : '-';
  const metaParts = [t.date];
  if (t.type === 'personal_expense') metaParts.push(t.category ? t.category : 'ยังไม่จัด', paymentLabel(t.paymentMethod, cards, cashAccounts));
  if (t.type === 'income') metaParts.push(t.category);
  if (t.type === 'pending_claim') metaParts.push(t.claimCategory, TYPE_LABEL[t.type]);
  if (t.type === 'loan_given') metaParts.push(t.status === 'repaid' ? 'คืนแล้ว' : 'ค้างคืน');
  return (
    <li className="pft-tx">
      {icon}
      <div className="pft-tx-info">
        <span className="pft-tx-desc">{t.description || TYPE_LABEL[t.type]}</span>
        <span className="pft-tx-meta">{metaParts.filter(Boolean).join(' · ')}</span>
      </div>
      <span className={`pft-tx-amt ${sign === '+' ? 'pft-green' : 'pft-red'}`}>{sign}฿{fmtMoney(t.amount)}</span>
      <button className="pft-del" onClick={() => onDelete(t.id)}><Trash2 size={14} /></button>
    </li>
  );
}
